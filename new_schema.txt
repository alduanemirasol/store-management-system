// Product Units: All valid packaging/measurement options per product
// Each row = one way to sell or restock a product (e.g. Rice by "kg" or "sack")

- product_id                FK → products.id, NOT NULL

- notes                     TEXT, NULL
- created_at                TIMESTAMP, DEFAULT CURRENT_TIMESTAMP

CONSTRAINT: Only one row per product_id can have is_default_selling = TRUE

- updated_at                TIMESTAMP, DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
CONSTRAINT: No two active rows (expiry_date IS NULL or >= today) for the same product_unit_id

// Pricing Tiers: Bulk, bundle, volume, and promotional pricing rules
// Replaces both the old product_unit_pricing_tiers AND promotions tables.
// tier_type controls how the price is applied:
//   BUNDLE_PRICE    = total_price is the fixed cost for exactly (qty_min = qty_max) units
//   VOLUME_DISCOUNT = price_per_unit replaces standard price for qty_min to qty_max units
//   FLAT_RATE       = price_per_unit is fixed regardless of standard price
pricing_tiers
- id                        PK, AUTO INCREMENT
- product_unit_id           FK → product_units.id, NOT NULL
- label                     VARCHAR(100), NULL               -- Optional promo name e.g. "New Year Sale"
- tier_type                 ENUM('BUNDLE_PRICE','VOLUME_DISCOUNT','FLAT_RATE'), NOT NULL
- quantity_min              INT UNSIGNED, NOT NULL           -- Min quantity to qualify
- quantity_max              INT UNSIGNED, NOT NULL           -- Max quantity (use 999999 for open-ended)
- price_per_unit            DECIMAL(10,2), NULL              -- Used for VOLUME_DISCOUNT, FLAT_RATE
- total_price               DECIMAL(10,2), NULL              -- Used for BUNDLE_PRICE
- effective_from            DATE, NOT NULL
- effective_to              DATE, NULL                       -- NULL = currently active
- created_by                FK → users.id, NOT NULL
- created_at                TIMESTAMP, DEFAULT CURRENT_TIMESTAMP
- updated_by                FK → users.id, NULL
- updated_at                TIMESTAMP, DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
CONSTRAINT: quantity_min <= quantity_max
CONSTRAINT: No overlapping (quantity_min, quantity_max) ranges for the same
            product_unit_id within the same active date window
CONSTRAINT: price_per_unit NOT NULL when tier_type IN ('VOLUME_DISCOUNT','FLAT_RATE')
CONSTRAINT: total_price NOT NULL when tier_type = 'BUNDLE_PRICE'

// EXAMPLES:
// Candy - 3 pcs = 5 PHP (bundle):
//   product_unit_id=candy_piece, tier_type='BUNDLE_PRICE',
//   quantity_min=3, quantity_max=3, total_price=5.00
//
// Candy - 10+ pcs = 1.50 PHP each (volume discount):
//   product_unit_id=candy_piece, tier_type='VOLUME_DISCOUNT',
//   quantity_min=10, quantity_max=999999, price_per_unit=1.50
//
// Rice - January promo, 54 PHP/kg:
//   product_unit_id=rice_kg, tier_type='FLAT_RATE', label='January Sale',
//   quantity_min=1, quantity_max=999999, price_per_unit=54.00,
//   effective_from='2024-01-01', effective_to='2024-01-31'

// Use Case: Check if a pricing tier applies for a given quantity
// SELECT * FROM pricing_tiers
// WHERE product_unit_id = ?
//   AND quantity_min <= ?          -- ? = qty being sold
//   AND quantity_max >= ?
//   AND effective_from <= CURDATE()
//   AND (effective_to IS NULL OR effective_to >= CURDATE())
// ORDER BY quantity_min DESC LIMIT 1;

// Pricing Resolution Logic (apply in this order):
// 1. Check pricing_tiers for a matching active tier → use tier price
// 2. If no tier → use product_unit_prices.selling_price
// 3. If cashier applies manual override → use sale_items.manual_price_per_unit

// ============================================================================
// KEY CONCEPTS FOR DEVELOPERS
// ============================================================================

// STOCK TRACKING
// Always store and update product_stock.quantity in the product's base unit.
// Convert to display units only in the application layer using product_units.pack_quantity.
// Example: 5 sacks of Rice = 5 * 50 = 250 kg stored in product_stock.

// UNIT CONVERSIONS
// No separate product_unit_conversions table needed.
// Conversion factor is product_units.pack_quantity (base units per pack).
// Example: Rice sack → pack_quantity=50 means 1 sack = 50 kg (base unit).

// PRICING RESOLUTION ORDER
// 1. Check pricing_tiers for an active tier matching product_unit_id + quantity + today's date.
// 2. If tier found and tier_type = BUNDLE_PRICE  → total_price / quantity = unit_price
// 3. If tier found and tier_type = VOLUME_DISCOUNT or FLAT_RATE → price_per_unit = unit_price
// 4. If no tier → use product_unit_prices.selling_price
// 5. If cashier applies override → use sale_items.unit_price (manual) and set is_manual_priced = TRUE

// SOFT DELETES
// Tables with is_deleted: users, products, customers
// Always filter WHERE is_deleted = FALSE in application queries.
// Never physically delete these rows to preserve historical records.

// DERIVED VALUES (do not store, compute at query time)
// sale total          → SUM(si.quantity * si.unit_price) WHERE sale_id = ?
// sale line_total     → si.quantity * si.unit_price
// cash balance        → opening_balance + SUM of cash_fund_transactions
// credit outstanding  → credit.amount_owed - credit.amount_paid

// AUDIT TRAIL
// All transactional tables include created_by (FK → users.id).
// Pricing and product tables include created_by + updated_by.
// Stock changes are fully logged in stock_movements.
// Price overrides are captured directly in sale_items (is_manual_priced + approved_by).
// Cash movements are logged in cash_fund_transactions.

// VARIABLE-WEIGHT ITEMS (e.g. Cabbage sold by piece)
// Use sale_items.weight_per_piece_kg to record actual weight at time of sale.
// Use product_units.approx_base_qty_per_piece for estimated stock deduction when
// exact weight is not entered by the cashier.
