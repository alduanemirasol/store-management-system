# HOW TO USE THIS FILE
# ─────────────────────────────────────────────────────────────────────────────
# Each table block contains:
#   PURPOSE      — what the table represents
#   COLUMNS      — field name | data type | constraints/notes
#   FOREIGN KEYS — field → table.field
#   RULES        — hard logic the application must always follow
#
# Cross-cutting rules that apply everywhere:
#   [SOFT DELETE]  products, customers, suppliers: use is_deleted = TRUE, never DELETE.
#                  Always add WHERE is_deleted = FALSE on active-record queries.
#   [NO TOTALS]    No derived totals are stored. Always compute at query time.
#   [BASE UNIT]    Stock is always stored in the product's base unit (kg, mL, piece).
#                  Convert to display units using product_units.pack_quantity.
#   [IMMUTABLE]    stock_movements and cash_fund_transactions rows are never
#                  updated or deleted after insert.
#   [CREDIT VIEW]  Always use the credit_balances VIEW for balances — never query
#                  credit.amount_paid (that column does not exist).

## TABLE: product_units
## PURPOSE: Every packaging or selling format for a product.
##          One row = one sellable or restockable unit option (e.g. "Rice by sack").
##          pack_quantity is the key: it converts display units → base units.
##          FORMULA: base_qty = quantity × pack_quantity
##          EXAMPLE: 3 sacks of Rice → 3 × 50 = 150 kg deducted from stock.
# ┌──────────────────────────┬──────────────┬───────────────────────────────────────────────┐
# │ column                   │ type         │ notes                                         │
# ├──────────────────────────┼──────────────┼───────────────────────────────────────────────┤
# │ id                       │ INT (PK)     │ auto-increment                                │
# │ product_id               │ INT (FK)     │ → products.id                                │
# │ unit_id                  │ INT (FK)     │ → units.id — underlying unit type             │
# │ display_name             │ VARCHAR      │ UI label — e.g. "sack", "250mL", "tray"       │
# │ pack_quantity            │ DECIMAL 12,4 │ base units per 1 display unit                 │
# │ is_default_selling       │ BOOLEAN      │ TRUE = pre-selected on POS (1 per product)    │
# │ can_restock              │ BOOLEAN      │ TRUE = available on restock screen             │
# │ can_sell                 │ BOOLEAN      │ TRUE = available on sales screen               │
# │ approx_base_qty_per_piece│ DECIMAL 10,4 │ nullable — kg/piece for variable-weight items │
# │ notes                    │ TEXT         │ nullable                                       │
# │ created_at               │ TIMESTAMP    │                                               │
# │ updated_at               │ TIMESTAMP    │                                               │
# └──────────────────────────┴──────────────┴───────────────────────────────────────────────┘
# FOREIGN KEYS: product_id → products.id
#               unit_id → units.id
# CONSTRAINT: Exactly ONE row per product may have is_default_selling = TRUE.
# REFERENCED BY: product_unit_prices.product_unit_id, pricing_tiers.product_unit_id,
#                sale_items.product_unit_id, stock_receipt_items.product_unit_id
#
# SAMPLE DATA:
#   product   display_name  pack_qty  default  can_restock  can_sell
#   Rice       kg            1        FALSE    TRUE         TRUE
#   Rice       sack          50       TRUE     TRUE         TRUE      ← 1 sack = 50 kg
#   Egg        piece         1        TRUE     TRUE         TRUE
#   Egg        tray          30       FALSE    TRUE         TRUE      ← 1 tray = 30 pcs
#   Oil        250mL         250      TRUE     FALSE        TRUE      ← sell only
#   Oil        container     20000    FALSE    TRUE         TRUE      ← 1 container = 20 L
#   Candy      piece         1        TRUE     TRUE         TRUE
#   Candy      pack          100      FALSE    TRUE         TRUE      ← 1 pack = 100 pcs
#   Cabbage    kg            1        FALSE    TRUE         TRUE
#   Cabbage    sack          25       FALSE    TRUE         FALSE     ← restock only
#   Cabbage    piece         ~0.5kg   TRUE     FALSE        TRUE      ← variable weight

## TABLE: product_unit_prices
## PURPOSE: Standard cost and selling price for each product unit.
##          Used as the fallback when no active pricing tier applies.
##          Only ONE active row per product_unit_id at a time
##          (active = expiry_date IS NULL or expiry_date >= today).
# ┌────────────────┬──────────────┬──────────────────────────────────────────────┐
# │ column         │ type         │ notes                                        │
# ├────────────────┼──────────────┼──────────────────────────────────────────────┤
# │ id             │ INT (PK)     │ auto-increment                               │
# │ product_unit_id│ INT (FK)     │ → product_units.id                          │
# │ purchase_price │ DECIMAL 10,2 │ cost paid to supplier                        │
# │ selling_price  │ DECIMAL 10,2 │ default retail price                         │
# │ effective_date │ DATE         │ when this price row becomes active           │
# │ expiry_date    │ DATE         │ nullable — NULL = currently active            │
# │ created_at     │ TIMESTAMP    │                                              │
# │ updated_at     │ TIMESTAMP    │                                              │
# └────────────────┴──────────────┴──────────────────────────────────────────────┘
# FOREIGN KEYS: product_unit_id → product_units.id
# CONSTRAINT: Partial unique index enforces one active row per product_unit_id:
#   UNIQUE INDEX uq_active_price ON product_unit_prices
#     (product_unit_id, (CASE WHEN expiry_date IS NULL THEN 1 END))
# AI NOTE: When inserting a new price, first set expiry_date on the current
#          active row, then insert the new row with expiry_date = NULL.
#
# SAMPLE DATA:
#   product unit     purchase   selling
#   Rice kg          55.00      57.00
#   Rice sack        2750.00    2850.00
#   Egg piece        9.00       9.00
#   Oil 250mL        22.50      37.00
#   Candy piece      1.50       2.00
#   Cabbage kg       180.50     200.00
#   Cabbage piece    0.00       25.00

---

## TABLE: pricing_tiers
## PURPOSE: Price overrides for bulk quantities, bundles, or promotional periods.
##          Takes priority over product_unit_prices when quantity and date match.
# ┌────────────────┬──────────────┬──────────────────────────────────────────────────┐
# │ column         │ type         │ notes                                            │
# ├────────────────┼──────────────┼──────────────────────────────────────────────────┤
# │ id             │ INT (PK)     │ auto-increment                                   │
# │ product_unit_id│ INT (FK)     │ → product_units.id                              │
# │ label          │ VARCHAR      │ nullable — promo name e.g. "January Sale"        │
# │ tier_type      │ ENUM         │ BUNDLE_PRICE | VOLUME_DISCOUNT | FLAT_RATE       │
# │ quantity_min   │ DECIMAL      │ minimum qty to qualify                           │
# │ quantity_max   │ DECIMAL      │ maximum qty — use 999999 for open-ended          │
# │ price_per_unit │ DECIMAL 10,2 │ unit price for VOLUME_DISCOUNT and FLAT_RATE     │
# │ total_price    │ DECIMAL 10,2 │ fixed bundle total for BUNDLE_PRICE              │
# │ effective_from │ DATE         │ promo start date                                 │
# │ effective_to   │ DATE         │ nullable — NULL = currently active               │
# │ created_at     │ TIMESTAMP    │                                                  │
# │ updated_at     │ TIMESTAMP    │                                                  │
# └────────────────┴──────────────┴──────────────────────────────────────────────────┘
# FOREIGN KEYS: product_unit_id → product_units.id
# CHECK CONSTRAINTS (DB-enforced):
#   quantity_min <= quantity_max
#   (VOLUME_DISCOUNT or FLAT_RATE) → price_per_unit must NOT be null
#   BUNDLE_PRICE                   → total_price must NOT be null
#
# TIER TYPE GUIDE:
#   BUNDLE_PRICE    → fixed total for an exact qty
#                     effective unit_price = total_price / quantity
#                     EXAMPLE: 3 Candy = 5 PHP total
#   VOLUME_DISCOUNT → per-unit price for a qty range
#                     effective unit_price = price_per_unit
#                     EXAMPLE: 10+ Candy = 1.50 PHP each
#   FLAT_RATE       → per-unit price regardless of standard price
#                     effective unit_price = price_per_unit
#                     EXAMPLE: Rice January sale = 54 PHP/kg
#
# PRICING RESOLUTION — always follow this order:
#   Step 1: Check pricing_tiers for an active tier matching the product_unit and qty.
#           WHERE product_unit_id = ?
#             AND quantity_min <= :qty AND quantity_max >= :qty
#             AND effective_from <= CURDATE()
#             AND (effective_to IS NULL OR effective_to >= CURDATE())
#   Step 2: If no tier found → use product_unit_prices.selling_price (active row).
#   Step 3: If owner manually overrides at POS → use sale_items.unit_price
#           and set is_manual_priced = TRUE + fill manual_price_reason.

## TABLE: stock_receipt_items
## PURPOSE: One row per product line within a stock receipt.
##          Triggers stock additions to product_stock and stock_movements.
# ┌──────────────────┬──────────────┬──────────────────────────────────────────────────┐
# │ column           │ type         │ notes                                            │
# ├──────────────────┼──────────────┼──────────────────────────────────────────────────┤
# │ id               │ INT (PK)     │ auto-increment                                   │
# │ stock_receipt_id │ INT (FK)     │ → stock_receipts.id                             │
# │ product_unit_id  │ INT (FK)     │ → product_units.id — unit the supplier delivered │
# │ quantity_received│ DECIMAL 12,4 │ in the display unit (e.g. sacks, trays)          │
# │ cost_per_unit    │ DECIMAL 10,2 │ actual price paid per display unit                │
# │ notes            │ VARCHAR 255  │ nullable                                          │
# │ created_at       │ TIMESTAMP    │                                                  │
# └──────────────────┴──────────────┴──────────────────────────────────────────────────┘
# FOREIGN KEYS: stock_receipt_id → stock_receipts.id
#               product_unit_id → product_units.id
#
# STOCK ADDITION LOGIC (run for every row inserted):
#   base_qty_to_add = quantity_received × product_units.pack_quantity
#   1. UPDATE product_stock SET quantity = quantity + base_qty_to_add
#      WHERE product_id = (SELECT product_id FROM product_units WHERE id = product_unit_id)
#   2. INSERT INTO stock_movements:
#        product_id          = (derived from product_unit)
#        stock_log_reason    = 'Purchase'
#        quantity_changed    = +base_qty_to_add
#        reference_type      = 'RECEIPT'
#        reference_id        = stock_receipt_id
#
# NOTE: cost_per_unit is the actual transaction cost.
#       product_unit_prices.purchase_price is the reference/standard cost — they may differ.

## TABLE: sale_items
## PURPOSE: One row per product line within a sale.
##          Holds the final charged price (post-tier and post-override).
##          No line_total stored — compute as quantity × unit_price.
# ┌─────────────────────┬──────────────┬─────────────────────────────────────────────────┐
# │ column              │ type         │ notes                                           │
# ├─────────────────────┼──────────────┼─────────────────────────────────────────────────┤
# │ id                  │ INT (PK)     │ auto-increment                                  │
# │ sale_id             │ INT (FK)     │ → sales.id                                     │
# │ product_unit_id     │ INT (FK)     │ → product_units.id — the unit sold              │
# │ quantity            │ DECIMAL 10,4 │ how many of the display unit were sold          │
# │ unit_price          │ DECIMAL 10,2 │ final price charged per unit                    │
# │ is_manual_priced    │ BOOLEAN      │ TRUE if owner overrode the system price         │
# │ manual_price_reason │ VARCHAR 255  │ nullable — required when is_manual_priced = TRUE│
# │ weight_per_piece_kg │ DECIMAL 10,4 │ nullable — actual weighed kg (variable items)   │
# │ created_at          │ TIMESTAMP    │                                                 │
# └─────────────────────┴──────────────┴─────────────────────────────────────────────────┘
# FOREIGN KEYS: sale_id → sales.id
#               product_unit_id → product_units.id
# REFERENCED BY: sales_returns.sale_item_id
#
# STOCK DEDUCTION LOGIC (run for every row inserted):
#   IF weight_per_piece_kg IS NOT NULL:
#     base_qty_to_deduct = weight_per_piece_kg   ← actual weighed amount
#   ELSE:
#     base_qty_to_deduct = quantity × product_units.pack_quantity
#   INSERT INTO stock_movements:
#     quantity_changed = -base_qty_to_deduct
#     reference_type   = 'SALE'
#     reference_id     = sale_id

---

## TABLE: sales_returns
## PURPOSE: Records a return (partial or full) against a previous sale.
##          Always triggers stock restoration via stock_movements.
# ┌───────────────────┬──────────────┬──────────────────────────────────────────────────┐
# │ column            │ type         │ notes                                            │
# ├───────────────────┼──────────────┼──────────────────────────────────────────────────┤
# │ id                │ INT (PK)     │ auto-increment                                   │
# │ sale_id           │ INT (FK)     │ → sales.id                                      │
# │ sale_item_id      │ INT (FK)     │ → sale_items.id — nullable (NULL = whole sale)   │

#   IF sale_item_id IS NOT NULL (partial return):

## TABLE: credit
## PURPOSE: Tracks that a sale was paid on credit. One row per credit sale.
##          Customer is NOT stored here — always derive via JOIN to sales.
##          Amount paid is NOT stored here — always compute via the credit_balances VIEW.
# ┌────────────┬──────────────┬────────────────────────────────────────────────────┐
# │ column     │ type         │ notes                                              │
# ├────────────┼──────────────┼────────────────────────────────────────────────────┤
# │ id         │ INT (PK)     │ auto-increment                                     │
# │ sale_id    │ INT (FK)     │ → sales.id — UNIQUE (one credit per sale)         │
# │ amount_owed│ DECIMAL 10,2 │ original credit amount                             │
# │ due_date   │ DATE         │ repayment deadline                                 │
# │ status     │ ENUM         │ PENDING | PARTIALLY_PAID | PAID | OVERDUE          │
# │ notes      │ TEXT         │ nullable                                            │
# │ created_at │ TIMESTAMP    │                                                    │
# │ updated_at │ TIMESTAMP    │                                                    │
# └────────────┴──────────────┴────────────────────────────────────────────────────┘
# FOREIGN KEYS: sale_id → sales.id (UNIQUE)
# REFERENCED BY: credit_payments.credit_id
#
# CREDIT LIMIT CHECK — run before approving any new credit sale:
#   SELECT c.credit_limit, COALESCE(SUM(cb.balance), 0) AS outstanding
#   FROM customers c
#   LEFT JOIN credit cr ON cr.sale_id IN (SELECT id FROM sales WHERE customer_id = c.id)
#   LEFT JOIN credit_balances cb ON cr.id = cb.credit_id AND cb.status != 'PAID'
#   WHERE c.id = ? AND c.is_deleted = FALSE
#   GROUP BY c.id;
#   → Approve only if (outstanding + new_sale_amount) <= credit_limit

---

## TABLE: credit_payments
## PURPOSE: Records one payment instalment against a credit record.
##          After each insert, recalculate and update credit.status.
# ┌────────────────┬──────────────┬──────────────────────────────────────────┐
# │ column         │ type         │ notes                                    │
# ├────────────────┼──────────────┼──────────────────────────────────────────┤
# │ id             │ INT (PK)     │ auto-increment                           │
# │ credit_id      │ INT (FK)     │ → credit.id                             │
# │ amount_paid    │ DECIMAL 10,2 │ payment amount                           │
# │ payment_method │ VARCHAR 50   │ e.g. Cash, GCash                         │
# │ paid_at        │ TIMESTAMP    │ when payment was received                 │
# │ notes          │ TEXT         │ nullable                                  │
# │ created_at     │ TIMESTAMP    │                                          │
# └────────────────┴──────────────┴──────────────────────────────────────────┘
# FOREIGN KEYS: credit_id → credit.id
#
# STATUS UPDATE LOGIC (after every insert):
#   balance = credit.amount_owed - SUM(credit_payments.amount_paid) for this credit_id
#   IF balance = 0              → status = 'PAID'
#   IF balance > 0 AND payments exist → status = 'PARTIALLY_PAID'
#   IF balance > 0 AND due_date < today AND no payments → status = 'OVERDUE'
#   ELSE                        → status = 'PENDING'

---

## VIEW: credit_balances
## PURPOSE: Always-accurate computed credit balance. Use this for all credit queries.
##          Never try to read credit.amount_paid — that column does not exist.
# COLUMNS RETURNED:
#   credit_id    — credit.id
#   customer_id  — derived from sales.customer_id
#   sale_id      — credit.sale_id
#   amount_owed  — credit.amount_owed
#   amount_paid  — SUM(credit_payments.amount_paid) — computed
#   balance      — amount_owed - amount_paid — computed
#   status       — credit.status (PENDING | PARTIALLY_PAID | PAID | OVERDUE)
#   due_date     — credit.due_date
#
# VIEW DEFINITION:
#   SELECT cr.id AS credit_id, s.customer_id, cr.sale_id, cr.amount_owed,
#          COALESCE(SUM(cp.amount_paid), 0) AS amount_paid,
#          cr.amount_owed - COALESCE(SUM(cp.amount_paid), 0) AS balance,
#          cr.status, cr.due_date
#   FROM credit cr
#   JOIN sales s ON cr.sale_id = s.id
#   LEFT JOIN credit_payments cp ON cr.id = cp.credit_id
#   GROUP BY cr.id, s.customer_id, cr.sale_id, cr.amount_owed, cr.status, cr.due_date;
#
# USE FOR: customer outstanding summary | credit limit checks | overdue detection | reports

## TABLE: cash_fund
## PURPOSE: Cash drawer for one business session.
##          Owner opens it at session start and closes it at end.
##          Current balance is always COMPUTED, never stored.
# ┌─────────────────┬──────────────┬────────────────────────────────────────────────┐
# │ column          │ type         │ notes                                          │
# ├─────────────────┼──────────────┼────────────────────────────────────────────────┤
# │ id              │ INT (PK)     │ auto-increment                                 │
# │ opening_balance │ DECIMAL 10,2 │ cash counted at session start                  │
# │ closed_at       │ TIMESTAMP    │ nullable — NULL means session is still open     │
# │ closing_balance │ DECIMAL 10,2 │ nullable — physical cash count at session close │
# │ notes           │ TEXT         │ nullable                                        │
# │ created_at      │ TIMESTAMP    │                                                │
# │ updated_at      │ TIMESTAMP    │                                                │
# └─────────────────┴──────────────┴────────────────────────────────────────────────┘
# FOREIGN KEYS: none
# REFERENCED BY: cash_fund_transactions.cash_fund_id
#
# BALANCE FORMULA:
#   current_balance = opening_balance + SUM(
#     CASE WHEN transaction_type = 'IN'  THEN amount
#          WHEN transaction_type = 'OUT' THEN -amount
#          ELSE amount  -- ADJUSTMENT (can be positive or negative)
#     END
#   ) FROM cash_fund_transactions WHERE cash_fund_id = ?

---

## TABLE: cash_fund_transactions
## PURPOSE: Immutable log of every cash movement within a session.
##          Never updated or deleted. Use ADJUSTMENT to fix errors.
# ┌──────────────────┬──────────────┬──────────────────────────────────────────────────┐
# │ column           │ type         │ notes                                            │
# ├──────────────────┼──────────────┼──────────────────────────────────────────────────┤
# │ id               │ INT (PK)     │ auto-increment                                   │
# │ cash_fund_id     │ INT (FK)     │ → cash_fund.id                                  │
# │ transaction_type │ ENUM         │ IN | OUT | ADJUSTMENT                            │
# │ amount           │ DECIMAL 10,2 │ always positive — direction set by type          │
# │ reference_type   │ VARCHAR      │ nullable — 'SALE' | 'EXPENSE' | 'RETURN'         │
# │ reference_id     │ INT          │ nullable — PK of the related record               │
# │ description      │ VARCHAR 255  │                                                  │
# │ created_at       │ TIMESTAMP    │                                                  │
# └──────────────────┴──────────────┴──────────────────────────────────────────────────┘
# FOREIGN KEYS: cash_fund_id → cash_fund.id
# RULE: NEVER update or delete rows. Use ADJUSTMENT type to correct mistakes.
# NOTE: reference_type + reference_id is polymorphic — no DB-level FK.
#       App must validate reference_id for the given reference_type before inserting.
#
# TRANSACTION TYPE GUIDE:
#   IN         → cash received (e.g. customer pays for a sale)
#   OUT        → cash paid out (e.g. expense or supplier payment)
#   ADJUSTMENT → manual correction after a physical cash count

################################################################################
# SECTION 14 — BUSINESS RULES QUICK REFERENCE
################################################################################
#
#  R1  STOCK IN BASE UNIT
#      product_stock always stores kg/mL/piece. Convert using pack_quantity.
#
#  R2  EVERY STOCK CHANGE = 1 stock_movements ROW
#      Covers: sales, returns, receipt items, damage, recount, adjustment.
#
#  R3  PRICING ORDER
#      pricing_tiers (if active + qty match) → product_unit_prices → manual override
#
#  R4  MANUAL PRICE OVERRIDE
#      Set is_manual_priced = TRUE and fill manual_price_reason.
#      No approval needed (single-owner system).
#
#  R5  CREDIT APPROVAL CHECKLIST
#      a. customer.is_deleted = FALSE
#      b. outstanding (from credit_balances VIEW) + new amount <= credit_limit
#      c. Insert credit row with sale_id only — customer derived via JOIN.
#
#  R6  FULL-SALE RETURN STOCK RESTORATION
#      Insert one stock_movements row per sale_item — never one aggregate row.
#
#  R7  VARIABLE-WEIGHT ITEMS
#      Use weight_per_piece_kg if weighed at POS.
#      Fall back to product_units.approx_base_qty_per_piece if not weighed.
#
#  R8  CASH TRANSACTIONS IMMUTABLE
#      Never UPDATE or DELETE cash_fund_transactions. Use ADJUSTMENT type to fix errors.
#
#  R9  SOFT DELETES
#      products, customers, suppliers: set is_deleted = TRUE, never DELETE.
#      All historical records (sales, movements, receipts) remain queryable.
#
#  R10 CREDIT BALANCE ALWAYS COMPUTED
#      Always use credit_balances VIEW. credit.amount_paid does not exist.
#
#  R11 PLANNED PURCHASES VIA stock_receipts
#      Use stock_receipts + stock_receipt_items for supplier deliveries.
#      Direct stock_movements with reason='Purchase' reserved for corrections only.
#
#  R12 POLYMORPHIC LINKS NEED APP VALIDATION
#      stock_movements.reference_id and cash_fund_transactions.reference_id
#      have no DB-level FK. App must validate before every insert.

# 15-B  PRICING
# ══════════════════════════════════════════════════════════════════════════════

# GET CURRENT PRICE FOR A PRODUCT UNIT
# Step 1 — check for an active tier matching the quantity
  SELECT tier_type, price_per_unit, total_price, label
  FROM pricing_tiers
  WHERE product_unit_id = :unit_id
    AND quantity_min <= :qty AND quantity_max >= :qty
    AND effective_from <= CURDATE()
    AND (effective_to IS NULL OR effective_to >= CURDATE())
  ORDER BY quantity_min DESC LIMIT 1;
# Step 2 — fallback to standard price if no tier found
  SELECT selling_price, purchase_price
  FROM product_unit_prices
  WHERE product_unit_id = :unit_id
    AND effective_date <= CURDATE()
    AND (expiry_date IS NULL OR expiry_date >= CURDATE())
  ORDER BY effective_date DESC LIMIT 1;

---

# GET ALL ACTIVE PRICES FOR EVERY UNIT OF A PRODUCT
  SELECT pu.display_name, pup.selling_price, pup.purchase_price, pup.effective_date
  FROM product_units pu
  JOIN product_unit_prices pup ON pu.id = pup.product_unit_id
  WHERE pu.product_id = :product_id
    AND pup.expiry_date IS NULL
  ORDER BY pu.display_name;

---

# GET ALL ACTIVE PRICING TIERS FOR A PRODUCT UNIT
  SELECT tier_type, label, quantity_min, quantity_max,
         price_per_unit, total_price, effective_from, effective_to
  FROM pricing_tiers
  WHERE product_unit_id = :unit_id
    AND effective_from <= CURDATE()
    AND (effective_to IS NULL OR effective_to >= CURDATE())
  ORDER BY quantity_min;

---

# GET PRICE HISTORY FOR A PRODUCT UNIT (all rows, newest first)
  SELECT selling_price, purchase_price, effective_date, expiry_date
  FROM product_unit_prices
  WHERE product_unit_id = :unit_id
  ORDER BY effective_date DESC;

---

# GET ALL CURRENTLY ACTIVE PROMOTIONS (any product)
  SELECT pt.label, pt.tier_type,
         p.name AS product, pu.display_name AS unit,
         pt.quantity_min, pt.quantity_max,
         pt.price_per_unit, pt.total_price,
         pt.effective_from, pt.effective_to
  FROM pricing_tiers pt

  WHERE pt.effective_from <= CURDATE()

# 15-D  SALES

# GET SALE HEADER + LINE ITEMS FOR ONE SALE
  SELECT s.id, s.sale_date, s.notes,
         COALESCE(CONCAT(c.first_name, ' ', c.last_name), 'Walk-in') AS customer,
         pt.name AS payment_method,
         p.name AS product, pu.display_name AS unit,
         si.quantity, si.unit_price,
         (si.quantity * si.unit_price) AS line_total,
         si.is_manual_priced, si.manual_price_reason,
         SUM(si.quantity * si.unit_price) OVER (PARTITION BY s.id) AS sale_total
  FROM sales s
  LEFT JOIN customers c ON s.customer_id = c.id
  JOIN payment_types pt ON s.payment_type_id = pt.id
  JOIN sale_items si ON s.id = si.sale_id
  JOIN product_units pu ON si.product_unit_id = pu.id
  JOIN products p ON pu.product_id = p.id
  WHERE s.id = :sale_id;

---

# GET DAILY SALES SUMMARY (today)
  SELECT COUNT(DISTINCT s.id)            AS total_transactions,
         SUM(si.quantity * si.unit_price) AS total_revenue
  FROM sales s
  JOIN sale_items si ON s.id = si.sale_id
  WHERE DATE(s.sale_date) = CURDATE();

---

# GET SALES SUMMARY FOR A DATE RANGE
  SELECT DATE(s.sale_date)               AS sale_date,
         COUNT(DISTINCT s.id)            AS total_transactions,
         SUM(si.quantity * si.unit_price) AS total_revenue
  FROM sales s
  JOIN sale_items si ON s.id = si.sale_id
  WHERE s.sale_date >= :date_from
    AND s.sale_date <  :date_to + INTERVAL 1 DAY
  GROUP BY DATE(s.sale_date)
  ORDER BY sale_date DESC;

---

# GET MONTHLY SALES SUMMARY (current year)
  SELECT MONTH(s.sale_date)              AS month,
         COUNT(DISTINCT s.id)            AS total_transactions,
         SUM(si.quantity * si.unit_price) AS total_revenue
  FROM sales s
  JOIN sale_items si ON s.id = si.sale_id
  WHERE YEAR(s.sale_date) = YEAR(CURDATE())
  GROUP BY MONTH(s.sale_date)
  ORDER BY month;

# GET ALL SALES FOR A CUSTOMER (newest first)
  SELECT s.id, s.sale_date, pt.name AS payment_method,
         SUM(si.quantity * si.unit_price) AS total
  FROM sales s
  JOIN payment_types pt ON s.payment_type_id = pt.id
  JOIN sale_items si ON s.id = si.sale_id
  WHERE s.customer_id = :customer_id
  GROUP BY s.id, s.sale_date, pt.name
  ORDER BY s.sale_date DESC;

---

# GET TOP-SELLING PRODUCTS BY REVENUE (date range)
  SELECT p.name AS product, pu.display_name AS unit,
         SUM(si.quantity)                   AS total_qty_sold,
         SUM(si.quantity * si.unit_price)   AS total_revenue
  FROM sale_items si
  JOIN product_units pu ON si.product_unit_id = pu.id
  JOIN products p ON pu.product_id = p.id
  JOIN sales s ON si.sale_id = s.id
  WHERE s.sale_date >= :date_from
    AND s.sale_date <  :date_to + INTERVAL 1 DAY
  GROUP BY p.id, p.name, pu.id, pu.display_name
  ORDER BY total_revenue DESC
  LIMIT 10;

# GET TOP-SELLING PRODUCTS BY QUANTITY SOLD (date range)
  SELECT p.name AS product, u.name AS base_unit,

  FROM sale_items si
  JOIN product_units pu ON si.product_unit_id = pu.id

# GET SALES BREAKDOWN BY PAYMENT METHOD (date range)
  SELECT pt.name AS payment_method,

# GET SALES BREAKDOWN BY CATEGORY (date range)

  JOIN product_units pu ON si.product_unit_id = pu.id

    AND s.sale_date <  :date_to + INTERVAL 1 DAY

# GET ALL MANUALLY OVERRIDDEN PRICES IN A DATE RANGE
  SELECT s.sale_date, s.id AS sale_id,

# GET ALL RETURNS FOR A DATE RANGE
  SELECT s.sale_date, sr.created_at AS returned_at,

  LEFT JOIN product_units pu ON si.product_unit_id = pu.id
  LEFT JOIN products p ON pu.product_id = p.id
  WHERE sr.created_at >= :date_from
    AND sr.created_at <  :date_to + INTERVAL 1 DAY
  ORDER BY sr.created_at DESC;

# GET TOTAL REFUNDS ISSUED IN A DATE RANGE
  SELECT COUNT(*) AS total_returns,
         SUM(sr.refund_amount) AS total_refunded
  FROM sales_returns sr
  WHERE sr.created_at >= :date_from
    AND sr.created_at <  :date_to + INTERVAL 1 DAY;

---

# GET GROSS PROFIT ESTIMATE FOR A DATE RANGE
# Uses product_unit_prices.purchase_price as cost reference.
# NOTE: actual cost may differ from receipt cost_per_unit.
  SELECT p.name AS product,
         SUM(si.quantity * si.unit_price)                            AS revenue,
         SUM(si.quantity * pu.pack_quantity * pup.purchase_price)    AS estimated_cost,
         SUM(si.quantity * si.unit_price)
           - SUM(si.quantity * pu.pack_quantity * pup.purchase_price) AS gross_profit
  FROM sale_items si
  JOIN sales s ON si.sale_id = s.id
  JOIN product_units pu ON si.product_unit_id = pu.id
  JOIN products p ON pu.product_id = p.id
  JOIN product_unit_prices pup
    ON pup.product_unit_id = pu.id
   AND pup.expiry_date IS NULL
  WHERE s.sale_date >= :date_from
    AND s.sale_date <  :date_to + INTERVAL 1 DAY
  GROUP BY p.id, p.name
  ORDER BY gross_profit DESC;

# CHECK CUSTOMER CREDIT LIMIT AVAILABILITY
# Returns current outstanding and how much headroom remains.
  SELECT c.credit_limit,
         COALESCE(SUM(cb.balance), 0)                   AS outstanding,
         c.credit_limit - COALESCE(SUM(cb.balance), 0)  AS available
  FROM customers c
  LEFT JOIN credit cr ON cr.sale_id IN (
    SELECT id FROM sales WHERE customer_id = c.id
  )
  LEFT JOIN credit_balances cb
    ON cr.id = cb.credit_id AND cb.status != 'PAID'
  WHERE c.id = :customer_id
    AND c.is_deleted = FALSE
  GROUP BY c.id, c.credit_limit;

# 15-I  CASH MANAGEMENT
# ══════════════════════════════════════════════════════════════════════════════

# GET THE CURRENTLY OPEN CASH SESSION (NULL if none open)
  SELECT id, opening_balance, created_at
  FROM cash_fund
  WHERE closed_at IS NULL
  ORDER BY created_at DESC
  LIMIT 1;

---

# GET CURRENT CASH BALANCE FOR AN OPEN SESSION
  SELECT cf.opening_balance,
         cf.opening_balance + COALESCE(SUM(
           CASE WHEN cft.transaction_type = 'IN'  THEN  cft.amount
                WHEN cft.transaction_type = 'OUT' THEN -cft.amount
                ELSE cft.amount END), 0) AS current_balance
  FROM cash_fund cf
  LEFT JOIN cash_fund_transactions cft ON cf.id = cft.cash_fund_id
  WHERE cf.id = :cash_fund_id
  GROUP BY cf.id, cf.opening_balance;

---

# GET TRANSACTION LOG FOR A SESSION (newest first)
  SELECT cft.created_at, cft.transaction_type, cft.amount,
         cft.reference_type, cft.reference_id, cft.description
  FROM cash_fund_transactions cft
  WHERE cft.cash_fund_id = :cash_fund_id
  ORDER BY cft.created_at DESC;

---

# GET RUNNING BALANCE THROUGH A SESSION (chronological)
  SELECT cft.created_at, cft.transaction_type, cft.amount, cft.description,
         cf.opening_balance + SUM(
           CASE WHEN cft.transaction_type = 'IN'  THEN  cft.amount
                WHEN cft.transaction_type = 'OUT' THEN -cft.amount
                ELSE cft.amount END
         ) OVER (ORDER BY cft.created_at ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
           AS running_balance
  FROM cash_fund_transactions cft
  JOIN cash_fund cf ON cft.cash_fund_id = cf.id
  WHERE cft.cash_fund_id = :cash_fund_id
  ORDER BY cft.created_at;

---

# GET SESSION CASH RECONCILIATION (closing summary)
  SELECT cf.opening_balance,
         cf.opening_balance + COALESCE(SUM(
           CASE WHEN cft.transaction_type = 'IN'  THEN  cft.amount
                WHEN cft.transaction_type = 'OUT' THEN -cft.amount
                ELSE cft.amount END), 0) AS computed_balance,
         cf.closing_balance,
         cf.closing_balance - (
           cf.opening_balance + COALESCE(SUM(
             CASE WHEN cft.transaction_type = 'IN'  THEN  cft.amount
                  WHEN cft.transaction_type = 'OUT' THEN -cft.amount
                  ELSE cft.amount END), 0)
         ) AS discrepancy
  FROM cash_fund cf
  LEFT JOIN cash_fund_transactions cft ON cf.id = cft.cash_fund_id
  WHERE cf.id = :cash_fund_id
  GROUP BY cf.id, cf.opening_balance, cf.closing_balance;

---

# GET CASH-IN TOTAL VS CASH-OUT TOTAL FOR A SESSION
  SELECT
    SUM(CASE WHEN transaction_type = 'IN'  THEN amount ELSE 0 END) AS total_in,
    SUM(CASE WHEN transaction_type = 'OUT' THEN amount ELSE 0 END) AS total_out,
    SUM(CASE WHEN transaction_type = 'IN'  THEN  amount
             WHEN transaction_type = 'OUT' THEN -amount
             ELSE amount END)                                        AS net_movement
  FROM cash_fund_transactions
  WHERE cash_fund_id = :cash_fund_id;

---

# GET SESSION HISTORY (all past closed sessions, newest first)
  SELECT cf.id, cf.created_at AS opened_at, cf.closed_at,
         cf.opening_balance, cf.closing_balance,
         cf.closing_balance - cf.opening_balance AS declared_movement
  FROM cash_fund cf
  WHERE cf.closed_at IS NOT NULL
  ORDER BY cf.closed_at DESC;

# 15-J  REPORTING & DASHBOARD
# ══════════════════════════════════════════════════════════════════════════════

# DAILY P&L SNAPSHOT (today)
# Revenue - expenses - refunds = net
  SELECT
    (SELECT COALESCE(SUM(si.quantity * si.unit_price), 0)
       FROM sale_items si
       JOIN sales s ON si.sale_id = s.id
      WHERE DATE(s.sale_date) = CURDATE())           AS gross_revenue,
    (SELECT COALESCE(SUM(amount), 0)
       FROM expenses
      WHERE expense_date = CURDATE())                AS total_expenses,
    (SELECT COALESCE(SUM(refund_amount), 0)
       FROM sales_returns sr
       JOIN sales s ON sr.sale_id = s.id
      WHERE DATE(s.sale_date) = CURDATE())           AS total_refunds;

---

# WEEKLY REVENUE TREND (last 7 days including today)
  SELECT DATE(s.sale_date)               AS day,
         COUNT(DISTINCT s.id)            AS transactions,
         SUM(si.quantity * si.unit_price) AS revenue
  FROM sales s
  JOIN sale_items si ON s.id = si.sale_id
  WHERE s.sale_date >= CURDATE() - INTERVAL 6 DAY
  GROUP BY DATE(s.sale_date)
  ORDER BY day;

---

# INVENTORY VALUE SNAPSHOT (cost × current stock)
  SELECT p.name AS product, u.name AS base_unit,
         ps.quantity AS stock_qty,
         pup.purchase_price AS unit_cost,
         (ps.quantity * pup.purchase_price) AS stock_value
  FROM product_stock ps
  JOIN products p ON ps.product_id = p.id
  JOIN units u ON p.base_unit_id = u.id
  JOIN product_units pu
    ON pu.product_id = p.id AND pu.pack_quantity = 1
  JOIN product_unit_prices pup
    ON pup.product_unit_id = pu.id AND pup.expiry_date IS NULL
  WHERE p.is_deleted = FALSE
  ORDER BY stock_value DESC;

---

# NET PROFIT ESTIMATE FOR A DATE RANGE
# Revenue minus purchasing cost of goods sold (based on receipt cost_per_unit)
  SELECT
    SUM(si.quantity * si.unit_price)                              AS total_revenue,
    (SELECT COALESCE(SUM(sri.quantity_received * sri.cost_per_unit), 0)
       FROM stock_receipt_items sri
       JOIN stock_receipts sr ON sri.stock_receipt_id = sr.id
      WHERE sr.receipt_date >= :date_from
        AND sr.receipt_date <= :date_to)                          AS total_cogs,
    (SELECT COALESCE(SUM(amount), 0)
       FROM expenses
      WHERE expense_date >= :date_from
        AND expense_date <= :date_to)                             AS total_expenses
  FROM sale_items si
  JOIN sales s ON si.sale_id = s.id
  WHERE s.sale_date >= :date_from
    AND s.sale_date <  :date_to + INTERVAL 1 DAY;

---

# PRODUCTS NOT SOLD IN THE LAST N DAYS (slow movers)
  SELECT p.id, p.name, c.name AS category,
         MAX(s.sale_date) AS last_sold_at,
         DATEDIFF(CURDATE(), MAX(s.sale_date)) AS days_since_last_sale
  FROM products p
  JOIN categories c ON p.category_id = c.id
  LEFT JOIN product_units pu ON pu.product_id = p.id
  LEFT JOIN sale_items si ON si.product_unit_id = pu.id
  LEFT JOIN sales s ON si.sale_id = s.id
  WHERE p.is_deleted = FALSE
  GROUP BY p.id, p.name, c.name
  HAVING days_since_last_sale > :days OR MAX(s.sale_date) IS NULL
  ORDER BY days_since_last_sale DESC;

---

# TOTAL OUTSTANDING CREDIT SUMMARY BY STATUS
  SELECT cb.status,
         COUNT(*)          AS record_count,
         SUM(cb.balance)   AS total_balance
  FROM credit_balances cb
  WHERE cb.status != 'PAID'
  GROUP BY cb.status
  ORDER BY total_balance DESC;
